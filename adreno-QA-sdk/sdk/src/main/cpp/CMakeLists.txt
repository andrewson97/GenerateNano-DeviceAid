cmake_minimum_required(VERSION 3.22.1)
project(adreno_gpu_llama)

# 🚀 COMPLETE QUALCOMM GGML-OPENCL COMPILATION SYSTEM

# Set up GGML source paths
set(GGML_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../adreno-llama-cpp/ggml)
set(GGML_SRC ${GGML_ROOT}/src)
set(OPENCL_SRC ${GGML_SRC}/ggml-opencl)

# Find pre-built llama libraries  
set(LLAMA_LIBS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI})

# Add imported libraries
add_library(llama SHARED IMPORTED)
set_target_properties(llama PROPERTIES IMPORTED_LOCATION ${LLAMA_LIBS_DIR}/libllama.so)

# Import GGML libraries (we'll build OpenCL from source)
add_library(ggml SHARED IMPORTED)
set_target_properties(ggml PROPERTIES IMPORTED_LOCATION ${LLAMA_LIBS_DIR}/libggml.so)

add_library(ggml-base SHARED IMPORTED)
set_target_properties(ggml-base PROPERTIES IMPORTED_LOCATION ${LLAMA_LIBS_DIR}/libggml-base.so)

# Add OpenCL library (for linking symbols, but excluded from packaging)
add_library(OpenCL SHARED IMPORTED)
set_target_properties(OpenCL PROPERTIES IMPORTED_LOCATION ${LLAMA_LIBS_DIR}/libOpenCL.so)

# Add OpenMP for threading
add_library(omp SHARED IMPORTED)
set_target_properties(omp PROPERTIES IMPORTED_LOCATION ${LLAMA_LIBS_DIR}/libomp.so)

# 🚀 ANDROID-COMPATIBLE GGML OPENCL BUILD SYSTEM
# Set OpenCL paths manually for Android (bypass find_package)
set(OpenCL_FOUND TRUE)
set(OpenCL_INCLUDE_DIRS ${CMAKE_CURRENT_SOURCE_DIR}/../../../../opencl-setup/OpenCL-Headers)
set(OpenCL_LIBRARIES ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libOpenCL.so)

# 🚀 COMPLETE QUALCOMM OPENCL COMPILE DEFINITIONS (from ggml-opencl.cpp analysis)
# These are CRITICAL for the 5000-line OpenCL implementation to work properly

# Core OpenCL backend definitions
add_compile_definitions(GGML_OPENCL_SOA_Q)                    # Struct-of-Arrays quantization layout
add_compile_definitions(GGML_OPENCL_TARGET_VERSION=300)       # OpenCL 3.0 API target  
add_compile_definitions(GGML_OPENCL_USE_ADRENO_KERNELS)       # Enable Adreno-specific optimizations
add_compile_definitions(GGML_OPENCL_EMBED_KERNELS)           # Use embedded kernel headers (.cl.h)

# OpenCL version definitions (required by ggml-opencl.cpp)
add_compile_definitions(CL_TARGET_OPENCL_VERSION=300)         # Match GGML_OPENCL_TARGET_VERSION
add_compile_definitions(CL_USE_DEPRECATED_OPENCL_1_2_APIS)    # For compatibility

# Backend system definitions
add_compile_definitions(GGML_USE_OPENCL)                      # Enable OpenCL in GGML
add_compile_definitions(GGML_BACKEND_OPENCL_DEVICE_COUNT=1)   # Single device support

# 🚀 GGML INTEGRATED APPROACH: Build OpenCL directly into main GGML library
# This follows GGML's architecture where backends are statically linked

# Add OpenCL source directly to the GGML library (not separate)
# This ensures proper backend registration during llama_backend_init()
set(OPENCL_SOURCES
    ${GGML_SRC}/ggml-opencl/ggml-opencl.cpp
)

# Handle kernel embedding (Qualcomm's approach adapted for Android)
find_program(Python3_EXECUTABLE python3 REQUIRED)
set(EMBED_KERNEL_SCRIPT "${GGML_SRC}/ggml-opencl/kernels/embed_kernel.py")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/autogenerated")
# Note: Will target our integrated library instead of separate TARGET_NAME

# Define kernel embedding function (from Qualcomm's CMakeLists.txt)
function(ggml_opencl_add_kernel KNAME)
    set(KERN_HDR ${CMAKE_CURRENT_BINARY_DIR}/autogenerated/${KNAME}.cl.h)
    set(KERN_SRC ${GGML_SRC}/ggml-opencl/kernels/${KNAME}.cl)

    message(STATUS "opencl: embedding kernel ${KNAME}")
    add_custom_command(
        OUTPUT ${KERN_HDR}
        COMMAND ${Python3_EXECUTABLE} ${EMBED_KERNEL_SCRIPT} ${KERN_SRC} ${KERN_HDR}
        DEPENDS ${KERN_SRC} ${EMBED_KERNEL_SCRIPT}
        COMMENT "Generate ${KERN_HDR}"
    )
    target_sources(adreno-llama-jni PRIVATE ${KERN_HDR})
endfunction()

# 🚀 CREATE JNI LIBRARY WITH COMPLETE QUALCOMM OPENCL BACKEND
add_library(adreno-llama-jni SHARED
    simple_llama_wrapper.cpp
    ${OPENCL_SOURCES}  # COMPLETE 5000-line ggml-opencl.cpp implementation
)

# Add all Qualcomm's kernels (complete list from their CMakeLists.txt)
set(GGML_OPENCL_KERNELS
    add clamp cpy cvt diag_mask_inf gelu gemv_noshuffle_general gemv_noshuffle
    get_rows im2col_f32 im2col_f16 mul_mat_Ab_Bi_8x4 mul_mv_f16_f16 mul_mv_f16_f32_1row
    mul_mv_f16_f32_l4 mul_mv_f16_f32 mul_mv_f32_f32 mul_mv_q4_0_f32 mul_mv_q4_0_f32_v
    mul_mv_q4_0_f32_8x_flat mul_mv_q4_0_f32_1d_8x_flat mul_mv_q4_0_f32_1d_16x_flat
    mul_mv_q6_k mul norm relu rms_norm rope scale silu softmax_4_f32 softmax_4_f16
    softmax_f32 softmax_f16 transpose
)

# NOW embed all kernels using Qualcomm's approach (target exists now)
foreach(K ${GGML_OPENCL_KERNELS})
    ggml_opencl_add_kernel(${K})
endforeach()

# Link with llama libraries + Qualcomm's OpenCL backend
target_link_libraries(adreno-llama-jni
    llama
    ggml
    ggml-base     # GGML base library
    OpenCL        # Link OpenCL symbols (but exclude from packaging - use system driver)
    omp           # OpenMP for threading
    android
    log
)

# 🚀 COMPLETE INCLUDE DIRECTORIES FOR QUALCOMM'S FULL OPENCL IMPLEMENTATION
# These paths are CRITICAL for the 5000-line ggml-opencl.cpp to compile properly
target_include_directories(adreno-llama-jni PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    
    # Core LLAMA.CPP includes
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../adreno-llama-cpp/include
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../adreno-llama-cpp/common  
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../adreno-llama-cpp/src
    
    # GGML Core includes (CRITICAL for backend integration)
    ${GGML_ROOT}/include                    # ggml.h, ggml-opencl.h
    ${GGML_SRC}                             # ggml-impl.h, ggml-backend.h  
    ${GGML_SRC}/ggml-opencl                 # ggml-opencl implementation
    ${CMAKE_CURRENT_BINARY_DIR}/autogenerated  # Generated .cl.h kernel headers
    
    # OpenCL Headers (REQUIRED for CL/cl.h, cl_platform.h, etc.)
    ${OpenCL_INCLUDE_DIRS}
)

# 🚀 ADDITIONAL COMPILE OPTIONS FOR COMPLETE OPENCL BACKEND
# Required for the complex ggml-opencl.cpp implementation
target_compile_options(adreno-llama-jni PRIVATE
    -Wno-unused-variable          # Suppress warnings in OpenCL code
    -Wno-unused-function          # Allow unused utility functions  
    -Wno-sign-compare             # Handle array index comparisons
    -Wno-missing-field-initializers  # OpenCL struct initialization
    -fno-strict-aliasing          # OpenCL memory access patterns
